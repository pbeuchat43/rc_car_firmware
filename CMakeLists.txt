cmake_minimum_required(VERSION 3.20)
option(RC_CAR_NATIVE_TESTS "Build portable tests with the host compiler" OFF)

if(RC_CAR_NATIVE_TESTS)
  project(rc_car_firmware_tests LANGUAGES CXX)
  enable_testing()
  add_executable(rc_car_firmware_tests tests/control_test.cpp)
  target_compile_features(rc_car_firmware_tests PRIVATE cxx_std_17)
  target_include_directories(rc_car_firmware_tests PRIVATE include)
  if(NOT MSVC)
    target_compile_options(rc_car_firmware_tests PRIVATE -Wall -Wextra -Werror)
  endif()
  add_test(NAME rc_car_firmware_tests COMMAND rc_car_firmware_tests "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
  return()
endif()

set(PICO_BOARD waveshare_rp2040_zero CACHE STRING "Pico board name")
if(NOT PICO_BOARD MATCHES "^waveshare_rp(2040|2350)_zero$")
  message(FATAL_ERROR "Select waveshare_rp2040_zero or waveshare_rp2350_zero")
endif()
if(NOT PICO_SDK_PATH AND DEFINED ENV{PICO_SDK_PATH})
  set(PICO_SDK_PATH "$ENV{PICO_SDK_PATH}")
endif()
if(NOT EXISTS "${PICO_SDK_PATH}/external/pico_sdk_import.cmake")
  message(FATAL_ERROR "Supply the provisioned Pico SDK path (-DPICO_SDK_PATH=...)")
endif()
include("${PICO_SDK_PATH}/external/pico_sdk_import.cmake")
project(rc_car_firmware LANGUAGES C CXX ASM)
pico_sdk_init()
if(NOT TARGET tinyusb_device)
  message(FATAL_ERROR "Initialize the Pico SDK lib/tinyusb submodule")
endif()
if(NOT EXISTS "${picotool_DIR}/picotoolConfig.cmake")
  message(FATAL_ERROR "Supply the provisioned picotool package directory (-Dpicotool_DIR=...)")
endif()
set(RC_CAR_PROFILE "" CACHE FILEPATH "Vehicle profile header")
if(NOT EXISTS "${RC_CAR_PROFILE}")
  message(FATAL_ERROR "Set -DRC_CAR_PROFILE to a bench-qualified profile header, or profiles/research_car.example.hpp for an unqualified build")
endif()

file(SHA256 "${RC_CAR_PROFILE}" profile_hash)
string(SUBSTRING "${profile_hash}" 0 16 RC_PROFILE_ID)
set(source_hashes "${profile_hash}/${PICO_BOARD}/${PICO_SDK_VERSION_STRING}")
foreach(input IN ITEMS CMakeLists.txt src/build_config.hpp.in src/main.cpp src/usb.cpp src/tusb_config.h
    include/rc_car/wire.hpp include/rc_car/control.hpp include/rc_car/device.hpp)
  file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/${input}" digest)
  string(APPEND source_hashes "/${digest}")
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${input}")
endforeach()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${RC_CAR_PROFILE}")
string(SHA256 build_hash "${source_hashes}")
string(SUBSTRING "${build_hash}" 0 16 RC_BUILD_ID)
configure_file(src/build_config.hpp.in generated/build_config.hpp @ONLY)
configure_file("${RC_CAR_PROFILE}" generated/vehicle_profile.hpp COPYONLY)
add_executable(rc_car_firmware src/main.cpp src/usb.cpp)
target_compile_features(rc_car_firmware PRIVATE cxx_std_17)
target_include_directories(rc_car_firmware PRIVATE include src "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_compile_options(rc_car_firmware PRIVATE -Wall -Wextra)
target_link_libraries(rc_car_firmware PRIVATE pico_stdlib pico_unique_id pico_rand hardware_pwm
    hardware_watchdog tinyusb_device)
pico_enable_stdio_usb(rc_car_firmware 0)
pico_enable_stdio_uart(rc_car_firmware 0)
pico_add_extra_outputs(rc_car_firmware)
add_custom_target(rc_car_firmware_bundle
  COMMAND "${CMAKE_COMMAND}" "-DUF2=${CMAKE_CURRENT_BINARY_DIR}/rc_car_firmware.uf2"
    "-DMANIFEST=${CMAKE_CURRENT_BINARY_DIR}/rc_car_firmware.manifest.json"
    "-DBOARD=${PICO_BOARD}" "-DBUILD=${RC_BUILD_ID}" "-DPROFILE=${RC_PROFILE_ID}"
    -P "${CMAKE_CURRENT_SOURCE_DIR}/cmake/artifact.cmake"
  DEPENDS rc_car_firmware VERBATIM)
